2. Processus automatique à la clôture
const cloturerAnnee = async (anneeId) => {
  await db.transaction(async (trx) => {

    // 1. Archiver l'année
    await trx('annees_academiques')
      .where({ id: anneeId })
      .update({ statut: 'archivee', cloturee_le: new Date() });

    // 2. Promouvoir les étudiants admis
    const admis = await trx('resultats_annuels')
      .where({ annee_id: anneeId, decision: 'admis' });

    for (const etudiant of admis) {
      const classeSuperieure = await getClasseSuperieure(etudiant.classe_id);
      await trx('students')
        .where({ id: etudiant.etudiant_id })
        .update({ classe_id: classeSuperieure.id });
    }

    // 3. Désactiver les étudiants diplômés
    await trx('students')
      .whereIn('id', etudiantsDiplomes)
      .update({ statut: 'diplome' });

    // 4. Créer la nouvelle année académique
    const nouvelleAnnee = await trx('annees_academiques').insert({
      nom: '2026-2027',
      statut: 'en_cours',
      date_debut: '2025-09-01'
    }).returning('*');

    // 5. Dupliquer la structure (classes, matières, UE)
    // sans les données (notes, absences, paiements)
    await dupliquerStructure(anneeId, nouvelleAnnee[0].id, trx);
  });
};
