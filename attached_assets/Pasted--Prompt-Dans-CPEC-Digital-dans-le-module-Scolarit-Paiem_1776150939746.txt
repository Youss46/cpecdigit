>
> #### 4.2 Validation automatique
>
> ```javascript
> // Vérifier que la somme des tranches = montant total
> const validerEcheancier = (tranches, montantTotal) => {
>   const sommeTranches = tranches
>     .reduce((sum, t) => sum + t.montant, 0);
>
>   if (sommeTranches !== montantTotal) {
>     return {
>       valide: false,
>       message: `La somme des tranches (${sommeTranches} FCFA)
>                 ne correspond pas au montant total
>                 (${montantTotal} FCFA)`
>     };
>   }
>   return { valide: true };
> };
> ```

> ### 5. Implémentation technique — Tables à créer
>
> ```sql
> -- Table échéanciers types par classe
> CREATE TABLE echeanciers (
>   id SERIAL PRIMARY KEY,
>   classe_id INTEGER REFERENCES classes(id),
>   annee_academique_id INTEGER,
>   montant_total INTEGER NOT NULL,
>   cree_le TIMESTAMP DEFAULT NOW()
> );
>
> -- Table tranches de l'échéancier
> CREATE TABLE echeancier_tranches (
>   id SERIAL PRIMARY KEY,
>   echeancier_id INTEGER REFERENCES echeanciers(id),
>   libelle VARCHAR(100) NOT NULL,  -- "1ère tranche", "Solde"...
>   montant INTEGER NOT NULL,
>   date_echeance DATE NOT NULL,
>   ordre INTEGER NOT NULL          -- 1, 2, 3...
> );
>
> -- Table suivi paiements par étudiant et tranche
> CREATE TABLE paiements_echeances (
>   id SERIAL PRIMARY KEY,
>   etudiant_id INTEGER REFERENCES students(id),
>   tranche_id INTEGER REFERENCES echeancier_tranches(id),
>   montant_paye INTEGER DEFAULT 0,
>   date_paiement DATE,
>   moyen_paiement VARCHAR(50),
>   statut VARCHAR(20) DEFAULT 'en_attente',
>   -- en_attente / partiel / paye / en_retard
>   cree_le TIMESTAMP DEFAULT NOW()
> );
> ```
>

> ### 6. Routes API à créer
>
> ```javascript
> // Créer un échéancier
> POST /api/echeanciers
>
> // Récupérer les échéanciers d'une classe
> GET /api/echeanciers?classe_id=X
>
> // Appliquer un échéancier à tous les étudiants d'une classe

>
> // Récupérer les tranches d'un étudiant
> GET /api/etudiants/:id/echeances
>
> // Enregistrer un paiement sur une tranche
> POST /api/paiements-echeances
> ```

Le point le plus important est la **validation que la somme des tranches = montant total** — sans ce contrôle, un échéancier mal configuré (ex: 3 tranches de 100 000 pour un total de 400 000) créerait des incohérences dans tout le suivi financier.
